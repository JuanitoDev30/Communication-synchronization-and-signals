169int main(int argc, char *argv[]) {
170 int socket_escucha;
171 int socket_cliente;
172 int *descriptor;
173 pthread_t hilo;
174 struct sigaction accion;
175
176 if (argc != 2 || !reconocer_politica(argv[1], &politica_actual)) {
177 fprintf(stderr, "Uso: %s sin|mutex|rwlock\n", argv[0]);
178 return EXIT_FAILURE;
179 }
180
181 /* Un cliente que cierra de golpe no debe terminar el servidor */
182 signal(SIGPIPE, SIG_IGN);
183
184 /* SA_RESTART queda fuera a proposito: accept debe retornar para que el ciclo vea la bandera */
185 memset(&accion, 0, sizeof(accion));
186 accion.sa_handler = al_interrumpir;
187 sigemptyset(&accion.sa_mask);
188 accion.sa_flags = 0;
189 if (sigaction(SIGINT, &accion, NULL) == -1) {
190 perror("sigaction");
191 return EXIT_FAILURE;
192 }
193
194 accion.sa_handler = al_alarmar;
195 if (sigaction(SIGALRM, &accion, NULL) == -1) {
196 perror("sigaction SIGALRM");
197 return EXIT_FAILURE;
198 }
199
200 memset(tablero, '.', TAM_TABLERO);
201 version_tablero = 0;
202
203 socket_escucha = crear_socket_escucha(PUERTO);
204 if (socket_escucha == -1) {
205 return EXIT_FAILURE;
206 }
207
208 printf("Tablero de %d casillas, politica %s, escuchando en el puerto %d\n",
209 TAM_TABLERO, argv[1], PUERTO);
210 fflush(stdout);
211
212 /* La primera alarma arranca una vez que el servidor esta listo para atender */
213 alarm(INTERVALO_ALARMA);
214
215 while (!terminar) {
216 if (alarma) {
217 char copia_tablero[TAM_TABLERO + 1];
218 int ver = leer_tablero(copia_tablero);
219 printf("[informe] version %d: %s\n", ver, copia_tablero);
220 fflush(stdout);
221 alarma = 0;
222 alarm(INTERVALO_ALARMA);
223 }
224
225 socket_cliente = accept(socket_escucha, NULL, NULL);
226 if (socket_cliente == -1) {
227 if (errno != EINTR) {
228 perror("accept");
229 }
230 continue;
231 }
232
233 /* El hilo recibe su propia copia del descriptor y la libera al terminar */
234 descriptor = malloc(sizeof(int));
235 if (descriptor == NULL) {
236 close(socket_cliente);
237 continue;
238 }
239 *descriptor = socket_cliente;
240
241 /* Bloquear SIGALRM antes de crear el hilo: el hijo hereda la mascara y nunca
242 recibe la senal, garantizando que los informes siempre lleguen a main */
243 {
244 sigset_t mascara;
245 sigemptyset(&mascara);
246 sigaddset(&mascara, SIGALRM);
247 pthread_sigmask(SIG_BLOCK, &mascara, NULL);
248
249 if (pthread_create(&hilo, NULL, atender_cliente, descriptor) != 0) {
250 perror("pthread_create");
251 pthread_sigmask(SIG_UNBLOCK, &mascara, NULL);
252 free(descriptor);
253 close(socket_cliente);
254 continue;
255 }
256
257 pthread_sigmask(SIG_UNBLOCK, &mascara, NULL);
258 }
259
260 /* Nadie espera el resultado del hilo: sus recursos se liberan al terminar */
261 pthread_detach(hilo);
262 }
263
264 close(socket_escucha);
265 {
266 int lecturas;
267 int escrituras;
268
269 obtener_contadores(&lecturas, &escrituras);
270 printf("Servidor detenido. Lecturas atendidas: %d, escrituras atendidas: %d\n", lecturas, escrituras);
271 }
272 return 0;
273}

291int crear_socket_escucha(int puerto) {
292 int socket_escucha;
293 int reutilizar = 1;
294 struct sockaddr_in direccion;
295
296 socket_escucha = socket(AF_INET, SOCK_STREAM, 0);
297 if (socket_escucha == -1) {
298 perror("socket");
299 return -1;
300 }
301
302 /* El puerto queda disponible de inmediato al terminar el servidor */
303 setsockopt(socket_escucha, SOL_SOCKET, SO_REUSEADDR, &reutilizar, sizeof(reutilizar));
304
305 memset(&direccion, 0, sizeof(direccion));
306 direccion.sin_family = AF_INET;
307 direccion.sin_addr.s_addr = htonl(INADDR_ANY);
308 direccion.sin_port = htons(puerto);
309
310 if (bind(socket_escucha, (struct sockaddr *) &direccion, sizeof(direccion)) == -1) {
311 perror("bind");
312 close(socket_escucha);
313 return -1;
314 }
315
316 if (listen(socket_escucha, 16) == -1) {
317 perror("listen");
318 close(socket_escucha);
319 return -1;
320 }
321
322 return socket_escucha;
323}

325void *atender_cliente(void *arg) {
326 int socket_cliente = *((int *) arg);
327 char orden[TAM_BUFFER];
328 char copia[TAM_TABLERO + 1];
329 char respuesta[TAM_BUFFER];
330 int version;
331 int longitud;

334
335 while (1) {
336 longitud = leer_linea(socket_cliente, orden, TAM_BUFFER);
337 if (longitud == -1) {
338 break;
339 }
340
341 if (strcmp(orden, "LEER") == 0) {
342 contar_operacion(0);
343 version = leer_tablero(copia);
344 snprintf(respuesta, TAM_BUFFER, "OK %d %s\n", version, copia);
345 } else if (strncmp(orden, "ESCRIBIR ", 9) == 0 && orden[9] != '\0') {
346 contar_operacion(1);
347 version = escribir_tablero(orden[9]);
348 snprintf(respuesta, TAM_BUFFER, "OK %d\n", version);
349 } else if (strcmp(orden, "SALIR") == 0) {
350 break;
351 } else {
352 snprintf(respuesta, TAM_BUFFER, "ERROR orden no reconocida\n");
353 }
354
355 if (write(socket_cliente, respuesta, strlen(respuesta)) == -1) {
356 break;
357 }
358 }
359
360 close(socket_cliente);

364int leer_linea(int socket_cliente, char *destino, int tam) {
365 int total = 0;
366 char caracter;
367 ssize_t leidos;
368
369 while (total < tam - 1) {
370 leidos = read(socket_cliente, &caracter, 1);
371
372 /* El cliente cerro la conexion: una linea a medias no se procesa */
373 if (leidos <= 0) {
374 return -1;
375 }
376 if (caracter == '\n') {
377 break;
378 }
379 if (caracter != '\r') {
380 destino[total] = caracter;
381 total++;
382 }
383 }
384
385 destino[total] = '\0';
386 return total;
387}

389int leer_tablero(char *destino) {
390 int i;
391 int version;
392
393 if (politica_actual == CON_MUTEX) {
394 pthread_mutex_lock(&candado);
395 } else if (politica_actual == CON_RWLOCK) {
396 pthread_rwlock_rdlock(&candado_rw);
397 }
398
399 /* La copia casilla por casilla amplia la ventana en la que otro hilo puede intervenir */
400 for (i = 0; i < TAM_TABLERO; i++) {
401 destino[i] = tablero[i];
402 usleep(RETARDO_LECTURA);
403 }
404 destino[TAM_TABLERO] = '\0';
405 version = version_tablero;
406
407 if (politica_actual == CON_MUTEX) {
408 pthread_mutex_unlock(&candado);
409 } else if (politica_actual == CON_RWLOCK) {
410 pthread_rwlock_unlock(&candado_rw);
411 }
412
413 return version;
