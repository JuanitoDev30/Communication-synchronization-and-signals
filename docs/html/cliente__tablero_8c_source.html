146int conectar(const char *direccion_ip, int puerto) {
147 int socket_servidor;
148 struct sockaddr_in direccion;
149
150 socket_servidor = socket(AF_INET, SOCK_STREAM, 0);
151 if (socket_servidor == -1) {
152 perror("socket");
153 return -1;
154 }
155
156 memset(&direccion, 0, sizeof(direccion));
157 direccion.sin_family = AF_INET;
158 direccion.sin_port = htons(puerto);
159
160 if (inet_pton(AF_INET, direccion_ip, &direccion.sin_addr) != 1) {
161 fprintf(stderr, "Direccion IPv4 no valida: %s\n", direccion_ip);
162 close(socket_servidor);
163 return -1;
164 }
165
166 if (connect(socket_servidor, (struct sockaddr *) &direccion, sizeof(direccion)) == -1) {
167 perror("connect");
168 close(socket_servidor);
169 return -1;
170 }
171
172 return socket_servidor;
173}

176 char linea[TAM_BUFFER];

191 leidos = read(socket_servidor, &caracter, 1);
192 if (leidos <= 0) {
193 return 0;
194 }
195 if (caracter == '\n') {
196 break;
197 }
198 respuesta[total] = caracter;
199 total++;
200 }
201
202 respuesta[total] = '\0';
203 return 1;
